generator client {
  provider = "prisma-client-py"
  interface = "asyncio"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// Modelo de Usuário
model User {
  id        String   @id @default(cuid())
  email     String   @unique
  name      String
  createdAt DateTime @default(now()) @map("created_at")
  updatedAt DateTime @updatedAt @map("updated_at")
  isActive  Boolean  @default(true) @map("is_active")

  // Relacionamentos
  emotions        UserEmotion[]
  preferences     UserPreference[]
  recommendations Recommendation[]
  feedbacks       UserFeedback[]

  @@map("users")
}

// Preferências Musicais do Usuário
model UserPreference {
  id              String @id @default(cuid())
  userId          String @map("user_id")
  genre           String
  preferenceLevel Int    @map("preference_level") // 1-10
  createdAt       DateTime @default(now()) @map("created_at")

  // Relacionamentos
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("user_preferences")
}

// Estados Emocionais do Usuário
model UserEmotion {
  id              String  @id @default(cuid())
  userId          String  @map("user_id")
  sessionId       String  @map("session_id")
  
  // Estados emocionais (escala 1-10)
  anxietyLevel    Int     @map("anxiety_level")
  depressionLevel Int     @map("depression_level")
  joyLevel        Int     @map("joy_level")
  stressLevel     Int     @map("stress_level")
  energyLevel     Int     @map("energy_level")
  
  // Contexto adicional
  context         String?
  situation       String?
  createdAt       DateTime @default(now()) @map("created_at")

  // Relacionamentos
  user            User @relation(fields: [userId], references: [id], onDelete: Cascade)
  recommendations Recommendation[]

  @@map("user_emotions")
}

// Músicas do Dataset
model Song {
  id              String  @id @default(cuid())
  title           String
  artistId        String  @map("artist_id")
  artistName      String? @map("artist_name")
  genre           String
  lyrics          String? @db.Text
  
  // Emoções originais (do dataset)
  anger           Float?
  fear            Float?
  joy             Float?
  sadness         Float?
  surprise        Float?
  
  // Características musicais (Spotify)
  valence         Float?  // Positividade musical
  energy          Float?  // Energia
  danceability    Float?  // Dançabilidade
  acousticness    Float?  // Acústica
  instrumentalness Float? // Instrumental
  liveness        Float?  // Vivacidade
  speechiness     Float?  // Fala
  tempo           Float?  // BPM
  
  // Características processadas pela IA
  relaxationScore Float?  @map("relaxation_score")
  therapeuticProfile String? @map("therapeutic_profile")
  
  createdAt       DateTime @default(now()) @map("created_at")

  // Relacionamentos
  recommendationSongs RecommendationSong[]
  feedbacks          UserFeedback[]

  @@map("songs")
}

// Recomendações Geradas
model Recommendation {
  id                String @id @default(cuid())
  userId            String @map("user_id")
  emotionRecordId   String @map("emotion_record_id")
  sessionId         String @map("session_id")
  
  // Metadados da recomendação
  recommendationType String @map("recommendation_type")
  therapyStrategy    String @map("therapy_strategy")
  
  createdAt         DateTime @default(now()) @map("created_at")

  // Relacionamentos
  user          User @relation(fields: [userId], references: [id], onDelete: Cascade)
  emotionRecord UserEmotion @relation(fields: [emotionRecordId], references: [id])
  songs         RecommendationSong[]
  feedbacks     UserFeedback[]

  @@map("recommendations")
}

// Músicas nas Recomendações
model RecommendationSong {
  id               String @id @default(cuid())
  recommendationId String @map("recommendation_id")
  songId           String @map("song_id")
  position         Int    // Posição na playlist
  compatibilityScore Float @map("compatibility_score")
  explanation      String? @db.Text
  
  createdAt        DateTime @default(now()) @map("created_at")

  // Relacionamentos
  recommendation Recommendation @relation(fields: [recommendationId], references: [id], onDelete: Cascade)
  song          Song @relation(fields: [songId], references: [id])

  @@map("recommendation_songs")
}

// Feedback do Usuário
model UserFeedback {
  id               String @id @default(cuid())
  userId           String @map("user_id")
  recommendationId String @map("recommendation_id")
  songId           String @map("song_id")
  
  // Feedback da música
  liked           Boolean?
  rating          Int?     // 1-5
  skipped         Boolean @default(false)
  playedDuration  Int?    @map("played_duration") // segundos
  
  // Estados emocionais pós-música (1-10)
  postAnxietyLevel     Int? @map("post_anxiety_level")
  postDepressionLevel  Int? @map("post_depression_level")
  postJoyLevel         Int? @map("post_joy_level")
  postStressLevel      Int? @map("post_stress_level")
  postEnergyLevel      Int? @map("post_energy_level")
  
  comments        String? @db.Text
  createdAt       DateTime @default(now()) @map("created_at")

  // Relacionamentos
  user           User @relation(fields: [userId], references: [id], onDelete: Cascade)
  recommendation Recommendation @relation(fields: [recommendationId], references: [id])
  song           Song @relation(fields: [songId], references: [id])

  @@map("user_feedback")
}

        description="Caminho para dataset com dados Spotify"

        default="data/emotion_music_data_original.csv",
        description="Caminho para dataset original BERT/GPT"

        """Configuração do Pydantic"""
        env_file = ".env"
        env_file_encoding = "utf-8"
        case_sensitive = False
        
        # Documentação dos campos
        schema_extra = {
            "example": {
                "database_url": "postgresql://sol_user:password@localhost:5432/sol_db",
                "secret_key": "your-super-secret-key-here-change-in-production",
                "debug": False,
                "log_level": "INFO"
            }
        }

# Instância global das configurações
settings = Settings()

# Validação de configurações críticas
def validate_settings() -> None:
    """
    Valida configurações críticas na inicialização
    Levanta exceções se alguma configuração obrigatória estiver ausente
    """
    
    required_settings = [
        ("database_url", "URL do banco de dados não configurada"),
        ("secret_key", "Chave secreta não configurada"),
        ("main_dataset_path", "Caminho do dataset principal não configurado")
    ]
    
    for setting_name, error_msg in required_settings:
        if not getattr(settings, setting_name, None):
            raise ValueError(f"{error_msg}. Configure a variável {setting_name.upper()}")
    
    # Validar se arquivos de dataset existem
    import os
    datasets = [
        settings.main_dataset_path,
        settings.spotify_dataset_path, 
        settings.original_dataset_path
    ]
    
    missing_files = [path for path in datasets if not os.path.exists(path)]
    if missing_files:
        print(f"⚠️  Arquivos de dataset não encontrados: {missing_files}")
        print("   O sistema funcionará, mas com funcionalidade limitada")

# Executar validação na importação
if __name__ != "__main__":
    validate_settings()

# app/database.py
"""
Configuração do Banco de Dados - Sistema SOL
Gerencia conexões com PostgreSQL usando Prisma
"""

import asyncio
import logging
from contextlib import asynccontextmanager
from typing import AsyncGenerator

from prisma import Prisma
from prisma.errors import PrismaError

from .config import settings

# Logger específico para database
logger = logging.getLogger(__name__)

class DatabaseManager:
    """
    Gerenciador de conexões com banco de dados
    Implementa padrão Singleton para uma única instância do Prisma
    """
    
    _instance = None
    _prisma: Prisma = None
    
    def __new__(cls):
        """Implementa padrão Singleton"""
        if cls._instance is None:
            cls._instance = super(DatabaseManager, cls).__new__(cls)
            cls._prisma = Prisma()
        return cls._instance
    
    async def connect(self) -> None:
        """
        Conecta ao banco de dados PostgreSQL
        
        Raises:
            PrismaError: Se não conseguir conectar ao banco
        """
        try:
            if not self._prisma.is_connected():
                logger.info("🔗 Conectando ao banco de dados PostgreSQL...")
                await self._prisma.connect()
                logger.info("✅ Conexão com banco estabelecida com sucesso")
            else:
                logger.info("📍 Banco já conectado")
                
        except PrismaError as e:
            logger.error(f"❌ Erro ao conectar com banco de dados: {e}")
            raise
        except Exception as e:
            logger.error(f"❌ Erro inesperado na conexão: {e}")
            raise
    
    async def disconnect(self) -> None:
        """
        Desconecta do banco de dados
        """
        try:
            if self._prisma.is_connected():
                logger.info("🔌 Desconectando do banco de dados...")
                await self._prisma.disconnect()
                logger.info("✅ Desconectado do banco com sucesso")
            else:
                logger.info("📍 Banco já desconectado")
                
        except PrismaError as e:
            logger.error(f"❌ Erro ao desconectar do banco: {e}")
        except Exception as e:
            logger.error(f"❌ Erro inesperado na desconexão: {e}")
    
    @property
    def client(self) -> Prisma:
        """
        Retorna a instância do cliente Prisma
        
        Returns:
            Prisma: Cliente para interação com banco
            
        Raises:
            RuntimeError: Se o banco não estiver conectado
        """
        if not self._prisma.is_connected():
            raise RuntimeError(
                "Banco de dados não conectado. "
                "Chame await database_manager.connect() primeiro"
            )
        return self._prisma
    
    async def health_check(self) -> bool:
        """
        Verifica se a conexão com banco está saudável
        
        Returns:
            bool: True se conexão estiver funcionando
        """
        try:
            # Tenta fazer uma query simples
            await self._prisma.query_raw("SELECT 1")
            return True
        except Exception as e:
            logger.error(f"❌ Health check do banco falhou: {e}")
            return False

# Instância global do gerenciador
database_manager = DatabaseManager()

@asynccontextmanager
async def get_database() -> AsyncGenerator[Prisma, None]:
    """
    Context manager para obter conexão com banco
    Garante que a conexão esteja ativa
    
    Usage:
        async with get_database() as db:
            users = await db.user.find_many()
            
    Yields:
        Prisma: Cliente conectado ao banco
    """
    try:
        await database_manager.connect()
        yield database_manager.client
    except Exception as e:
        logger.error(f"❌ Erro no context manager do banco: {e}")
        raise
    finally:
        # Não desconecta aqui, mantém conexão viva
        pass

# Dependency para FastAPI
async def get_db_dependency() -> Prisma:
    """
    Dependency do FastAPI para injeção do cliente do banco
    
    Usage:
        @app.get("/users")
        async def get_users(db: Prisma = Depends(get_db_dependency)):
            return await db.user.find_many()
    
    Returns:
        Prisma: Cliente conectado ao banco
    """
    await database_manager.connect()
    return database_manager.client

# Funções utilitárias para inicialização
async def init_database() -> None:
    """
    Inicializa o banco de dados
    Conecta e executa migrações se necessário
    """
    logger.info("🚀 Inicializando banco de dados...")
    
    try:
        await database_manager.connect()
        
        # Verificar se as tabelas existem
        health_ok = await database_manager.health_check()
        if health_ok:
            logger.info("✅ Banco de dados inicializado com sucesso")
        else:
            logger.warning("⚠️  Banco conectado, mas health check falhou")
            
    except Exception as e:
        logger.error(f"❌ Falha na inicialização do banco: {e}")
        raise

async def close_database() -> None:
    """
    Fecha conexões com banco de dados
    Chamado no shutdown da aplicação
    """
    logger.info("🔒 Fechando conexões com banco de dados...")
    await database_manager.disconnect()

# Exemplo de uso direto (para scripts)
async def example_usage():
    """
    Exemplo de como usar o DatabaseManager diretamente
    """
    try:
        # Conectar
        await database_manager.connect()
        
        # Usar
        db = database_manager.client
        users = await db.user.find_many()
        print(f"Encontrados {len(users)} usuários")
        
        # Health check
        is_healthy = await database_manager.health_check()
        print(f"Banco saudável: {is_healthy}")
        
    except Exception as e:
        print(f"Erro: {e}")
    finally:
        # Desconectar
        await database_manager.disconnect()

if __name__ == "__main__":
    # Executar exemplo se chamado diretamente
    asyncio.run(example_usage())
